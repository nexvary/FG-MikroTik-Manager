cmake_minimum_required(VERSION 3.20)
project(FgMtmDesktop VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Quick QuickControls2 Network Test)
qt_add_executable(fg_mtm WIN32 src/main.cpp src/Bridge.cpp src/Bridge.hpp src/Protocol.cpp src/Protocol.hpp)
qt_add_resources(fg_mtm "ui" PREFIX "/" FILES qml/Main.qml qml/Theme.qml qml/FgButton.qml qml/FgField.qml qml/NavIcon.qml packaging/fg-mtm.png)
target_link_libraries(fg_mtm PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Network)
if(WIN32)
 configure_file(packaging/app.rc ${CMAKE_CURRENT_BINARY_DIR}/app.rc @ONLY)
 target_sources(fg_mtm PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/app.rc)
endif()
enable_testing()
qt_add_executable(protocol_tests tests/test_protocol.cpp src/Protocol.cpp src/Protocol.hpp)
target_link_libraries(protocol_tests PRIVATE Qt6::Test Qt6::Network)
add_test(NAME protocol COMMAND protocol_tests)
